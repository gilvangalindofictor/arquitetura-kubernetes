## 🔍 Como Validar Este Status

```bash
# Verificar versão EKS
aws eks describe-cluster --name k8s-platform-prod --region us-east-1 \
  --query 'cluster.version' --output text
# Expected: 1.34

# Verificar versão nodes
kubectl get nodes -o wide | awk '{print $1, $5}'
# Expected: v1.31.13-eks-ecaa3a6 (pending upgrade)

# Verificar node groups
for ng in system workloads critical; do
  aws eks describe-nodegroup --cluster-name k8s-platform-prod --nodegroup-name $ng \
    --query 'nodegroup.{Name:nodegroupName,Desired:scalingConfig.desiredSize,Version:version}' \
    --output table
done

# Verificar EventBridge rules RDS
aws events list-rules --region us-east-1 \
  --query "Rules[?contains(Name, 'shutdown')].{Name:Name,Schedule:ScheduleExpression,State:State}" \
  --output table

# Verificar pods críticos
kubectl get pods -n gitlab-staging
kubectl get pods -n keycloak
kubectl get pods -n vault-system
kubectl get pods -n monitoring | grep -E "prometheus|grafana|loki|tempo"
```
